-- Se precisar recriar do zero, rode manualmente:  DROP DATABASE radiance;

CREATE DATABASE IF NOT EXISTS radiance

    COLLATE utf8mb4_unicode_ci;

USE radiance;

SET FOREIGN_KEY_CHECKS = 0;

-- =====================================================================
-- NOTA SOBRE SOFT DELETE + UNIQUE
-- ---------------------------------------------------------------------
-- No MySQL, NULL nunca e igual a NULL em indices UNIQUE. Entao um indice
-- UNIQUE(email, deleted_at) NAO impede dois registros ativos com o mesmo
-- email, porque ambos teriam deleted_at = NULL e seriam considerados
-- distintos.
--
-- Solucao usada aqui: coluna gerada `del_flag`, que vale 0 quando o
-- registro esta ativo e o timestamp da exclusao quando esta deletado.
-- Assim:
--   - dois registros ativos  -> del_flag = 0 nos dois -> COLISAO (bloqueia)
--   - registros deletados    -> del_flag diferente    -> permitido
-- =====================================================================


-- =====================================================================
-- 1. USERS
-- =====================================================================
CREATE TABLE users (
    id              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    name            VARCHAR(150)    NOT NULL,
    email           VARCHAR(150)    NOT NULL,
    password_hash   VARCHAR(255)    NOT NULL,
    cpf             VARCHAR(14)     NULL,
    phone           VARCHAR(20)     NULL,
    role            ENUM('customer','admin') NOT NULL DEFAULT 'customer',
    email_verified_at DATETIME      NULL,
    remember_token  VARCHAR(100)    NULL,
    last_login_at   DATETIME        NULL,

    created_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at      DATETIME        NULL,
    del_flag        BIGINT          AS (IFNULL(UNIX_TIMESTAMP(deleted_at),0)) STORED,

    PRIMARY KEY (id),
    UNIQUE KEY uq_users_email (email, del_flag),
    UNIQUE KEY uq_users_cpf   (cpf,   del_flag),
    KEY idx_users_role       (role),
    KEY idx_users_deleted_at (deleted_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- =====================================================================
-- 2. ADDRESSES

-- =====================================================================
-- 3. CATEGORIES (auto-relacionamento para subcategorias)
-- =====================================================================
CREATE TABLE categories (

    name            VARCHAR(100)    NOT NULL,
    slug            VARCHAR(120)    NOT NULL,

    image_path      VARCHAR(255)    NULL,
    position        INT             NOT NULL DEFAULT 0,
    active          TINYINT(1)      NOT NULL DEFAULT 1,

    created_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at      DATETIME        NULL,
    del_flag        BIGINT          AS (IFNULL(UNIX_TIMESTAMP(deleted_at),0)) STORED,

    PRIMARY KEY (id),
    UNIQUE KEY uq_cat_slug   (slug, del_flag),
    KEY idx_cat_parent       (parent_id),
    KEY idx_cat_active       (active),

) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- =====================================================================
-- 4. PRODUCTS
-- =====================================================================
CREATE TABLE products (
    id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    category_id       BIGINT UNSIGNED NOT NULL,
    name              VARCHAR(255)    NOT NULL,
    slug              VARCHAR(280)    NOT NULL,
    sku               VARCHAR(50)     NOT NULL,
    short_description VARCHAR(255)    NULL,
    description       TEXT            NULL,
    price             DECIMAL(10,2)   NOT NULL,
    promo_price       DECIMAL(10,2)   NULL,
    weight_grams      INT UNSIGNED    NOT NULL DEFAULT 0,
    active            TINYINT(1)      NOT NULL DEFAULT 1,
    featured          TINYINT(1)      NOT NULL DEFAULT 0,

    created_at        DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at        DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at        DATETIME        NULL,

        REFERENCES categories(id) ON DELETE RESTRICT ON UPDATE CASCADE,

    CONSTRAINT chk_prod_promo CHECK (promo_price IS NULL OR promo_price >= 0)

-- =====================================================================
-- 7. CARTS (anonimo por session_token OU logado por user_id)
-- =====================================================================
CREATE TABLE carts (
    id              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    user_id         BIGINT UNSIGNED NULL,
    session_token   CHAR(36)        NULL,
    status          ENUM('active','converted','abandoned') NOT NULL DEFAULT 'active',
    expires_at      DATETIME        NULL,

    created_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at      DATETIME        NULL,
    del_flag        BIGINT          AS (IFNULL(UNIX_TIMESTAMP(deleted_at),0)) STORED,

    PRIMARY KEY (id),
    UNIQUE KEY uq_cart_user    (user_id, status, del_flag),
    UNIQUE KEY uq_cart_session (session_token, del_flag),
    KEY idx_cart_status        (status),
    KEY idx_cart_expires       (expires_at),
    KEY idx_cart_deleted_at    (deleted_at),
    CONSTRAINT fk_cart_user FOREIGN KEY (user_id)
        REFERENCES users(id) ON DELETE RESTRICT ON UPDATE CASCADE
    -- REGRA DE NEGOCIO: user_id OU session_token deve estar preenchido.
    -- O MySQL rejeita CHECK nessa forma (erro 1901), entao a validacao
    -- fica na APLICACAO. Nunca insira os dois como NULL.
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- =====================================================================
-- 8. CART_ITEMS (preco congelado na adicao)

-- =====================================================================
-- 11. COUPONS  (criado antes de ORDERS por causa da FK)
-- =====================================================================
CREATE TABLE coupons (

    code              VARCHAR(50)     NOT NULL,
    description       VARCHAR(255)    NULL,
    type              ENUM('percent','fixed') NOT NULL,
    value             DECIMAL(10,2)   NOT NULL,
    min_order_value   DECIMAL(10,2)   NOT NULL DEFAULT 0.00,

    max_uses_per_user INT UNSIGNED    NOT NULL DEFAULT 1,
    used_count        INT UNSIGNED    NOT NULL DEFAULT 0,
    starts_at         DATETIME        NULL,
    expires_at        DATETIME        NULL,
    active            TINYINT(1)      NOT NULL DEFAULT 1,

    created_at        DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at        DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at        DATETIME        NULL,
    del_flag          BIGINT          AS (IFNULL(UNIX_TIMESTAMP(deleted_at),0)) STORED,

    PRIMARY KEY (id),
    UNIQUE KEY uq_coupon_code (code, del_flag),
    KEY idx_coupon_active     (active),
    KEY idx_coupon_expires    (expires_at),
    KEY idx_coupon_deleted_at (deleted_at),
    CONSTRAINT chk_coupon_value CHECK (value > 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- =====================================================================
-- 13. ORDERS
-- =====================================================================
CREATE TABLE orders (
    id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    order_number      VARCHAR(20)     NOT NULL,
    user_id           BIGINT UNSIGNED NOT NULL,
    status            ENUM('pendente','aguardando_pagamento','pago','separando',
                           'enviado','entregue','cancelado','expirado','estornado')
                      NOT NULL DEFAULT 'pendente',

    subtotal          DECIMAL(10,2)   NOT NULL DEFAULT 0.00,
    discount_total    DECIMAL(10,2)   NOT NULL DEFAULT 0.00,
    shipping_total    DECIMAL(10,2)   NOT NULL DEFAULT 0.00,
    total             DECIMAL(10,2)   NOT NULL DEFAULT 0.00,

    coupon_id         BIGINT UNSIGNED NULL,
    coupon_code       VARCHAR(50)     NULL COMMENT 'Snapshot',

    shipping_method   VARCHAR(50)     NULL,
    shipping_days     INT             NULL,
    tracking_code     VARCHAR(50)     NULL,

    customer_snapshot JSON            NULL COMMENT 'nome, email, cpf, telefone',
    shipping_snapshot JSON            NULL COMMENT 'endereco completo congelado',
    notes             TEXT            NULL,

    paid_at           DATETIME        NULL,
    shipped_at        DATETIME        NULL,
    delivered_at      DATETIME        NULL,
    cancelled_at      DATETIME        NULL,

    created_at        DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at        DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at        DATETIME        NULL,
    del_flag          BIGINT          AS (IFNULL(UNIX_TIMESTAMP(deleted_at),0)) STORED,

    PRIMARY KEY (id),
    UNIQUE KEY uq_order_number (order_number, del_flag),
    KEY idx_order_user       (user_id),
    KEY idx_order_status     (status),
    KEY idx_order_created    (created_at),
    KEY idx_order_deleted_at (deleted_at),
    KEY idx_order_painel     (status, created_at),
    CONSTRAINT fk_order_user FOREIGN KEY (user_id)
        REFERENCES users(id) ON DELETE RESTRICT ON UPDATE CASCADE,
    CONSTRAINT fk_order_coupon FOREIGN KEY (coupon_id)
        REFERENCES coupons(id) ON DELETE RESTRICT ON UPDATE CASCADE,
    CONSTRAINT chk_order_total CHECK (total >= 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- =====================================================================
-- VIEW: estoque disponivel (ja descontando reservas ativas)
-- =====================================================================
CREATE OR REPLACE VIEW vw_estoque_disponivel AS
SELECT
    v.id                AS variant_id,
    v.product_id,
    v.variant_value,
    v.stock_quantity,
    COALESCE(SUM(r.quantity), 0) AS reservado,
    v.stock_quantity - COALESCE(SUM(r.quantity), 0) AS disponivel
FROM product_variants v
LEFT JOIN stock_reservations r
       ON r.variant_id = v.id
      AND r.status     = 'active'
      AND r.expires_at > NOW()
      AND r.deleted_at IS NULL
WHERE v.deleted_at IS NULL
  AND v.active = 1
GROUP BY v.id, v.product_id, v.variant_value, v.stock_quantity;


-- =====================================================================
-- VIEW: produtos visiveis na loja
-- =====================================================================
CREATE OR REPLACE VIEW vw_produtos_publicos AS
SELECT
    p.id,
    p.name,
    p.slug,
    p.sku,
    p.short_description,
    p.price,
    p.promo_price,
    p.featured,
    p.category_id,
    c.name AS categoria,
    c.slug AS categoria_slug,
    (SELECT i.path
       FROM product_images i
      WHERE i.product_id = p.id
        AND i.deleted_at IS NULL
      ORDER BY i.is_main DESC, i.position ASC
      LIMIT 1) AS imagem_principal,
    COALESCE(est.estoque_total, 0) AS estoque_total
FROM products p
INNER JOIN categories c
        ON c.id = p.category_id
       AND c.deleted_at IS NULL
       AND c.active = 1
LEFT JOIN (
    SELECT product_id, SUM(disponivel) AS estoque_total
      FROM vw_estoque_disponivel
     GROUP BY product_id
) est ON est.product_id = p.id
WHERE p.deleted_at IS NULL
  AND p.active = 1
  AND COALESCE(est.estoque_total, 0) > 0;
